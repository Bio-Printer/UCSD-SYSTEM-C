# MAKEFILE -- the operating system in C: SYS-C:SYSTEM.C, the same as the one on
# BIGGY byte for byte.  From the shell: MAKE -f SYS-C:MAKEFILE (UCSD-C: and TOOLS:
# on line); MAKE -f UCSD-C:MAKEFILE runs it too.  SYSC.OBJ goes onto SYS-C:; MAKE -B
# ... compiles it again.
S = SYS-C:

SYS-C:SYSTEM.C : $(S)SYSC.OBJ
	$(CC) -O $(S) -L SYS-C:SYSTEM.C.=SYSC

$(S)SYSC.OBJ : $(S)SYSC.C $(S)GLOBALS.H $(S)SEG0.C $(S)SEGS.C
	$(CC) -O $(S) -C $(S)SYSC
